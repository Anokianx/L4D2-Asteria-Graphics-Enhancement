cmake_minimum_required(VERSION 3.20)
project(AsteriaOfficialFGCoreProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_definitions(UNICODE _UNICODE _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
set(FFX_PLATFORM_NAME x86)
set(FFX_FI ON)
set(FFX_OF ON)
set(FFX_BUILD_AS_DLL OFF)
set(FFX_INCLUDE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/sdk-v1.1.4/sdk/include")
set(FFX_SHARED_PATH "${CMAKE_CURRENT_SOURCE_DIR}/sdk-v1.1.4/sdk/src/shared")
set(FFX_HOST_PATH "${FFX_INCLUDE_PATH}/FidelityFX/host")
set(FFX_COMPONENTS_PATH "${CMAKE_CURRENT_SOURCE_DIR}/sdk-v1.1.4/sdk/src/components")
file(GLOB FFX_SHARED_SOURCES "${FFX_SHARED_PATH}/*.cpp" "${FFX_SHARED_PATH}/*.h")
add_subdirectory("${FFX_COMPONENTS_PATH}/frameinterpolation" frameinterpolation)
add_subdirectory("${FFX_COMPONENTS_PATH}/opticalflow" opticalflow)
add_executable(AsteriaOfficialFGCoreProbe core_probe.cpp)
target_link_libraries(AsteriaOfficialFGCoreProbe PRIVATE ffx_frameinterpolation_x86 ffx_opticalflow_x86)
